# Firmware component — Main.cpp entry point
set(EXTRA_REQS "")

if(NOT "$ENV{FLXOS_HEADLESS_MODE_ENABLED}" STREQUAL "ON")
    message("FIRMWARE: GRAPHICAL MODE DETECTED (via ENV)")
    list(APPEND EXTRA_REQS UI Applications)
else()
    message("FIRMWARE: HEADLESS MODE DETECTED (via ENV)")
endif()

# Register Firmware as a component
idf_component_register(
    SRCS "Source/Main.cpp"
    INCLUDE_DIRS ""
    REQUIRES ${EXTRA_REQS} Profiles HalModule System Connectivity Apps Services Kernel Core
)

# Inject LovyanGFX dependency into lvgl component
# distinct from standard requirements because we can't modify lvgl's CMakeLists.txt
if(NOT "$ENV{FLXOS_HEADLESS_MODE_ENABLED}" STREQUAL "ON" AND CONFIG_LV_USE_LOVYAN_GFX)
    idf_component_get_property(lvgl_lib lvgl COMPONENT_LIB)
    target_link_libraries(${lvgl_lib} PUBLIC idf::LovyanGFX)
    # Profiles provides Config.hpp (generated by profile.cmake)
    target_link_libraries(${lvgl_lib} PUBLIC idf::Profiles)
endif()

if(NOT "$ENV{FLXOS_HEADLESS_MODE_ENABLED}" STREQUAL "ON" AND CONFIG_LV_OS_IDLE_PERCENT_CUSTOM)
    idf_component_get_property(lvgl_lib lvgl COMPONENT_LIB)
    target_sources(${lvgl_lib} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../HalModule/Source/lv_os_custom.c")
endif()
